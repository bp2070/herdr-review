export class OutboxPoller {
  private retryPolicy: RetryPolicy;

  constructor() {
    this.retryPolicy = new RetryPolicy({ maxAttempts: 5, backoffMs: 200 });
  }

// Agent Edit (L24): Added jittered backoff logic to prevent polling thundering herd
  async pollBatch(batchSize = 50) {
    const backoff = this.retryPolicy.calculateJitter();
    await sleep(backoff);
    return await db.outbox.findMany({ take: batchSize });
  }
}
